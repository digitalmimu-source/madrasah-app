// ========================================================
// KODE GOOGLE APPS SCRIPT v2.5 - DATABASE ABSENSI FULL CLOUD
// ========================================================

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = e.parameter.action;
    var result = { success: false };
    
    var sheetStudents = getOrCreateSheet(ss, "Siswa", ["ID", "NISN", "Nama", "Kelas", "Gender"]);
    var sheetAttendance = getOrCreateSheet(ss, "Absensi", ["ID", "StudentID", "NISN", "Nama", "Kelas", "Tanggal", "Waktu", "Status"]);
    var sheetSettings = getOrCreateSheet(ss, "Pengaturan", ["Key", "Value"]);

    if (action === "get_all") {
      result = {
        success: true,
        students: getSheetDataAsObjects(sheetStudents),
        attendance: getSheetDataAsObjects(sheetAttendance),
        settings: getSettingsAsObject(sheetSettings)
      };
    } 
    else if (action === "sync_students" && e.postData) {
      var body = JSON.parse(e.postData.contents);
      saveObjectsToSheet(sheetStudents, body.students, ["id", "nisn", "name", "class", "gender"]);
      result = { success: true, message: "Data siswa berhasil disimpan!" };
    }
    else if (action === "sync_attendance" && e.postData) {
      var body = JSON.parse(e.postData.contents);
      saveObjectsToSheet(sheetAttendance, body.attendance, ["id", "studentId", "nisn", "name", "class", "date", "time", "status"]);
      result = { success: true, message: "Log absensi berhasil disimpan!" };
    }
    else if (action === "update_attendance" && e.postData) {
      var body = JSON.parse(e.postData.contents);
      updateRowInSheet(sheetAttendance, body.item, "id");
      result = { success: true, message: "Absensi berhasil diperbarui!" };
    }
    else if (action === "delete_attendance" && e.postData) {
      var body = JSON.parse(e.postData.contents);
      deleteRowFromSheet(sheetAttendance, body.id, "id");
      result = { success: true, message: "Absensi berhasil dihapus!" };
    }
    else if (action === "sync_settings" && e.postData) {
      var body = JSON.parse(e.postData.contents);
      saveSettingsFromObject(sheetSettings, body.settings);
      result = { success: true, message: "Pengaturan berhasil disimpan!" };
    }

    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, sheetName, headers) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#dcfce7");
  }
  return sheet;
}

function getSheetDataAsObjects(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0].map(function(h){ return h.toString().toLowerCase(); });
  var list = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      var key = headers[j] === "studentid" ? "studentId" : headers[j];
      obj[key] = String(data[i][j]);
    }
    list.push(obj);
  }
  return list;
}

function saveObjectsToSheet(sheet, items, keys) {
  sheet.clearContents();
  var headers = keys.map(function(k){ return k.toUpperCase(); });
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#dcfce7");
  
  if (!items || items.length === 0) return;
  var rows = items.map(function(item){
    return keys.map(function(k){ return item[k] || ""; });
  });
  sheet.getRange(2, 1, rows.length, keys.length).setValues(rows);
}

function updateRowInSheet(sheet, item, keyName) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;
  var headers = data[0].map(function(h){ return h.toString().toLowerCase(); });
  var keyIdx = headers.indexOf(keyName.toLowerCase());
  
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][keyIdx]) === String(item[keyName])) {
      sheet.getRange(i + 1, 6).setValue(item.date);
      sheet.getRange(i + 1, 7).setValue(item.time);
      sheet.getRange(i + 1, 8).setValue(item.status);
      break;
    }
  }
}

function deleteRowFromSheet(sheet, idValue, keyName) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;
  var headers = data[0].map(function(h){ return h.toString().toLowerCase(); });
  var keyIdx = headers.indexOf(keyName.toLowerCase());
  
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][keyIdx]) === String(idValue)) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
}

function getSettingsAsObject(sheet) {
  var data = sheet.getDataRange().getValues();
  var settings = {};
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) settings[data[i][0]] = String(data[i][1]);
  }
  return settings;
}

function saveSettingsFromObject(sheet, settingsObj) {
  sheet.clearContents();
  sheet.appendRow(["KEY", "VALUE"]);
  sheet.getRange(1, 1, 1, 2).setFontWeight("bold").setBackground("#dcfce7");
  var rows = [];
  for (var key in settingsObj) {
    rows.push([key, settingsObj[key]]);
  }
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, 2).setValues(rows);
  }
}
